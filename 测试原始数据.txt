
DR模式需要的额外配置
DR必须配置
# 1. 在 loopback 上添加 VIP
sudo ip addr add 192.168.72.160/32 dev lo

# 2. 禁止 RS 响应 VIP 的 ARP 请求（避免 ARP 冲突）
sudo sysctl -w net.ipv4.conf.all.arp_ignore=1
sudo sysctl -w net.ipv4.conf.all.arp_announce=2
sudo sysctl -w net.ipv4.conf.lo.arp_ignore=1
sudo sysctl -w net.ipv4.conf.lo.arp_announce=2

# 3. 验证
ip addr show lo
# 应该看到 192.168.72.160/32

如果要修改为FULLNAT，重启虚拟机最好
# 移除 loopback 上的 VIP
sudo ip addr del 192.168.72.160/32 dev lo

# 恢复 ARP 参数
sudo sysctl -w net.ipv4.conf.all.arp_ignore=0
sudo sysctl -w net.ipv4.conf.all.arp_announce=0

# 验证
ip addr show lo
